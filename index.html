function pesquisar(){ let texto = document.getElementById('pesquisa').value.toLowerCase(); // filtra categorias document.querySelectorAll('.categoria').forEach(cat=>{ let nome = cat.innerText.toLowerCase(); cat.style.display = nome.includes(texto) ? 'block' : 'none'; }); // filtra produtos também document.querySelectorAll('.produto').forEach(prod=>{ let nome = prod.innerText.toLowerCase(); prod.style.display = nome.includes(texto) ? 'block' : 'none'; }); } document.getElementById('pesquisa').addEventListener('input', pesquisar);
